// 1. تحويل تقييم السنتباون (Centipawns) إلى نسبة فوز لوجستية function cpToWinRate(cp) { return 1 / (1 + Math.pow(10, -cp / 400)); } // 2. دالة حساب الدقة المستندة إلى منهجية حجم الضرر (CAPS Logic) function calculatePlayerAccuracy(moveEvaluations) { let totalScore = 0; let totalMoves = moveEvaluations.length; if (totalMoves === 0) return 100.0; for (let i = 0; i < totalMoves; i++) { let evalBefore = moveEvaluations[i].before; let evalAfter = moveEvaluations[i].after; let winRateBefore = cpToWinRate(evalBefore); let winRateAfter = cpToWinRate(evalAfter); // حساب التراجع الحقيقي في فرصة الفوز لهذه النقلة let winRateDrop = Math.max(0, winRateBefore - winRateAfter); // تحويل الهبوط إلى نسبة دقة (كلما زاد الخطأ، انخفضت النسبة بشكل أسي طبيعي) let moveAccuracy = Math.exp(-4 * winRateDrop) * 100; totalScore += moveAccuracy; } let finalAccuracy = totalScore / totalMoves; return Math.max(0, Math.min(100, finalAccuracy)); } // 3. دالة تحليل المباراة واستخراج مصفوفات التقييم لكل من الأبيض والأسود async function analyzeGameWithStockfish(gamePgn, stockfishWorker) { // ملاحظة: تأكد من تهيئة رقعة الشطرنج ومكتبة chess.js هنا لتقسيم النقلات const chess = new Chess(); chess.loadPgn(gamePgn); const history = chess.history({ verbose: true }); // إعادة تعيين الرقعة للبداية للتحليل التسلسلي chess.reset(); let whiteEvaluations = []; let blackEvaluations = []; let lastEval = 0; // تقييم البداية متعادل for (let i = 0; i < history.length; i++) { let move = history[i]; let isWhite = (move.color === 'w'); // تنفيذ النقلة على الرقعة الوهمية chess.move(move); // طلب تقييم الموقف من محرك Stockfish (يُفضل ضبط العمق على 12 أو أكثر لضمان دقة الأخطاء) let currentEval = await getStockfishEvaluation(chess.fen(), stockfishWorker, 12); // توحيد اتجاه التقييم (دائماً من منظور اللاعب الذي قام بالنقلة الحالية) let evaluationForPlayer = isWhite ? currentEval : -currentEval; let evaluationBeforePlayer = isWhite ? lastEval : -lastEval; // تخزين التقييم قبل وبعد النقلة ضمن مصفوفة اللاعب المعني if (isWhite) { whiteEvaluations.push({ before: evaluationBeforePlayer, after: evaluationForPlayer }); } else { blackEvaluations.push({ before: evaluationBeforePlayer, after: evaluationForPlayer }); } lastEval = currentEval; } // حساب الدقة النهائية باستخدام المعاملات الحقيقية let whiteAcc = calculatePlayerAccuracy(whiteEvaluations); let blackAcc = calculatePlayerAccuracy(blackEvaluations); return { whiteAccuracy: whiteAcc.toFixed(1), // إرجاع الرقم برقم عشري واحد (مثلاً 78.3) blackAccuracy: blackAcc.toFixed(1) // إرجاع الرقم برقم عشري واحد (مثلاً 73.3) }; } // دالة مساعدة للتعامل مع رسائل محرك Stockfish (تأكد من مطابقتها لتهيئة الـ Worker في مشروعك) function getStockfishEvaluation(fen, worker, depth) { return new Promise((resolve) => { let currentEval = 0; function listener(event) { let message = event.data; // البحث عن تقييم السنتباون (cp) أو مات (mate) في مخرجات المحرك if (message.includes("infopoint") || message.includes("score")) { if (message.includes("cp")) { let match = message.match(/cp\s(-?\d+)/); if (match) currentEval = parseInt(match[1]); } else if (message.includes("mate")) { let match = message.match(/mate\s(-?\d+)/); if (match) { let mateIn = parseInt(match[1]); // تحويل كش المات إلى تقييم رقمي ضخم ومستقر currentEval = mateIn > 0 ? 10000 - (mateIn * 100) : -10000 - (mateIn * 100); } } } if (message.includes(`depth ${depth}`) || message.includes("bestmove")) { worker.removeEventListener('message', listener); resolve(currentEval); } } worker.addEventListener('message', listener); worker.postMessage(`position fen ${fen}`); worker.postMessage(`go depth ${depth}`); }); }
